Python:
import os
from mistralai import Mistral

# Initialize the client
api_key = os.environ.get("MISTRAL_API_KEY")
client = Mistral(api_key)
inputs = [
  {
    "role": "user",
    "content": "Hello there!",
  }
]
try:
    # Make the API call
    response = client.beta.conversations.start_stream(
        inputs=inputs,
        agent_id="ag_019a48a53a3f72359ae98b1de158b4e3",
    )
    # Process the stream
    for chunk in response:
        print(chunk)
except Exception as e:
    print(f"Error: {str(e)}")

  agentId: "ag_019a48a53a3f72359ae98b1de158b4e3",
  inputs: "Hello there!",
});
for await (const chunk of response) {
  console.log(chunk);
