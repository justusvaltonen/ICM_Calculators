Monte Carlo Simulation for n > 8
------------------------------
function approximateICM(stacks, payouts, samples = 100000) {
    const icmValues = new Array(stacks.length).fill(0);
    for (let i = 0; i < samples; i++) {
        const permutation = generateRandomPermutation(stacks.length);
        const probability = calculatePermutationProbability(permutation, stacks);
        for (let pos = 0< pos < stacks.length; pos++) {
            icmValues[permutation[pos]] += probability * payouts[pos];
        }
    }
    return icmValues.map(v => v / samples);
}
